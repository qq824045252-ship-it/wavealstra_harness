export type BotLifecycleStatus = "active" | "paused" | "disabled";

export type WorkMode = "once" | "watch";

export type WorkLifecycle = "active" | "paused" | "closing" | "closed";

export type WorkClosureConclusion = "succeeded" | "cancelled" | "ended_unsuccessfully";

export type WorkCatchUpPolicy = "latest" | "each" | "skip";

export type WorkParticipantRole = "collaborator" | "worker" | "watcher";

export type DeliverableStatus = "proposed" | "accepted" | "rejected" | "superseded";

/**
 * 专家模板版本快照 (对应设计文档第 11 节 & BW-A07)
 * 运行中执行固定绑定 snapshotVersion；模板更新进入 pendingVersion，需显式采纳才升为 activeVersion
 */
export interface ExpertProfileSnapshotDto {
  expertId: string;
  desiredVersion: number;
  activeVersion: number;
  pendingVersion?: number;
  systemPromptHash: string;
  allowedMcpIds: string[];
  allowedSkillIds: string[];
  capturedAt: number;
}

/**
 * 记忆分层与共享策略 (对应 F12 & BW-A08)

 */
export interface BotMemoryScopeDto {
  personalMemoryNamespace: string;
  sharePersonalMemoryWithTeam: boolean;
  allowUnconfirmedSpeculationInMemory: boolean;
}

export interface BotChannelBindingDto {
  channelType: "desktop" | "mobile" | "remote" | "gateway";
  channelIdentifier: string;

  boundAt: number;
}

/**

export interface BotDto {

  foregroundAllowedTools: string[];

  updatedAt: number;

/**
 * 验收条件 (兼容 shared/conversation-goal.ts 的 GoalCriterionDto 并扩展版本核验与实测标记，对应 BW-A34)
 */
export interface WorkCriterionDto {
  id: string;
  text: string;
  status: "pending" | "satisfied" | "failed" | "blocked";
  evidence?: string;
  /** 要求满足该条件的目标要求版本 (防止旧版本实测反馈自动通过新版本要求，对应 BW-A34) */
  requiredIntentRevision: number;
  /** 实际验证通过时对应的要求版本 */
  verifiedIntentRevision?: number;
  /** 是否要求真实设备/环境实测反馈 (如 Omarchy 实测，模型不得自行猜测满足) */
  requiresRealEnvironmentCheck?: boolean;
}

/**
 * 授权快照 (对应设计文档第 8 节)
 */
export interface AuthorizationSnapshotDto {
  allowedScope: string[];
  targetWhitelist: string[];
  allowGitBranchOrWorktree: boolean; // 对应 BW-A32 仓库禁止新分支/worktree 约束
  budgetLimitTokens?: number;
  expiresAt?: number;
  allowRedelegation: boolean;
}

/**
 * 预算窗口与消耗记账 (对应 F10 & BW-A40)

export interface WorkBudgetDto {

  maxTokens: number;
  usedTokens: number;

/**
 * 在 AppDatabase.ts 中执行 v27 -> v28 的事务迁移函数
 * 自动保证默认主 Bot 存在，并将旧 conversation_goals 平滑升级为独立 Work (F02)
 */
export function migrateToV28BotWorkPhase1(
  db: Database.Database,
  defaultBotId = "bot-mybuddy-primary",
  defaultUserId = "user-local-primary",
): void {
  const migrateTx = db.transaction(() => {
    db.exec(BOT_WORK_PHASE1_DDL);

    const now = Date.now();
    const defaultExpertSnapshot: ExpertProfileSnapshotDto = {
      expertId: "expert-default-coordinator",
      desiredVersion: 1,
      activeVersion: 1,
      systemPromptHash: "default-v1",
      allowedMcpIds: [],
      allowedSkillIds: [],
      capturedAt: now,
    };
    const defaultMemoryScope: BotMemoryScopeDto = {
      personalMemoryNamespace: `mem-${defaultBotId}`,
      sharePersonalMemoryWithTeam: false,
      allowUnconfirmedSpeculationInMemory: false,
    };

    db.prepare(`
      INSERT OR IGNORE INTO bots (
        bot_id, user_id, display_name, responsibility_summary,
        expert_profile_json, memory_scope_json, foreground_allowed_tools_json,
        notification_policy_json, status, created_at, updated_at
      ) VALUES (
        @botId, @userId, @displayName, @responsibilitySummary,
        @expertProfileJson, @memoryScopeJson, @foregroundAllowedToolsJson,
        @notificationPolicyJson, 'active', @now, @now
      )
    `).run({
      botId: defaultBotId,
      userId: defaultUserId,
      displayName: "MyBuddy",
      responsibilitySummary: "默认长期主负责人 Bot",
      expertProfileJson: JSON.stringify(defaultExpertSnapshot),
      memoryScopeJson: JSON.stringify(defaultMemoryScope),
      foregroundAllowedToolsJson: JSON.stringify([
        "work_query_status",
        "work_create",
        "work_amend",
        "work_pause",
        "work_resume",
        "work_cancel",
        "memory_search",
      ]),
      notificationPolicyJson: JSON.stringify({ inAppCard: true }),
      now,
    });

    // 检查是否存在旧表 conversation_goals，若存在则原位升级到 works 与 work_intent_snapshots
    const hasLegacyGoalsTable = db
      .prepare("SELECT name FROM sqlite_master WHERE type='table' AND name='conversation_goals'")
      .get() as { name: string } | undefined;

    if (hasLegacyGoalsTable) {
      type LegacyGoalRow = {
        id: string;
        task_id: string;
        title: string;
        description: string | null;
        status: "active" | "blocked" | "paused" | "achieved" | "cancelled";
        criteria_json: string;
        progress_summary: string | null;
        version: number;
        created_by: "user" | "agent";
        created_at: number;
        updated_at: number;
        achieved_at: number | null;
      };

      const legacyGoals = db
        .prepare("SELECT * FROM conversation_goals ORDER BY created_at ASC")
        .all() as LegacyGoalRow[];

      const insertWorkStmt = db.prepare(`
        INSERT OR IGNORE INTO works (
          work_id, user_id, legacy_task_id, owner_bot_id, owner_epoch,
          intent_revision, title, goal_description, constraints_json, criteria_json,
          authorization_json, active_intent_snapshot_id, mode, lifecycle,
          closure_conclusion, progress_summary, state_version, budget_json,
          created_by, created_at, updated_at, closed_at
        ) VALUES (
          @workId, @userId, @legacyTaskId, @ownerBotId, 1,
          @intentRevision, @title, @goalDescription, '[]', @criteriaJson,
          @authorizationJson, @snapshotId, 'once', @lifecycle,
          @closureConclusion, @progressSummary, @stateVersion, @budgetJson,
          @createdBy, @createdAt, @updatedAt, @closedAt
        )
      `);

      const insertSnapshotStmt = db.prepare(`
        INSERT OR IGNORE INTO work_intent_snapshots (
          snapshot_id, work_id, intent_revision, title, goal_description,
          constraints_json, criteria_json, authorization_json, expert_snapshot_json,
          created_by, created_at
        ) VALUES (
          @snapshotId, @workId, @intentRevision, @title, @goalDescription,
          '[]', @criteriaJson, @authorizationJson, @expertSnapshotJson,
          @createdBy, @createdAt
        )
      `);

      const insertConvLinkStmt = db.prepare(`
        INSERT OR IGNORE INTO work_conversations (
          work_id, conversation_id, channel_type, last_referenced_at
        ) VALUES (@workId, @conversationId, 'desktop', @updatedAt)
      `);

      for (const goal of legacyGoals) {
        const rawCriteria = JSON.parse(goal.criteria_json || "[]") as Array<{
          id: string;
          text: string;
          status?: string;
          evidence?: string;
        }>;
        const upgradedCriteria: WorkCriterionDto[] = rawCriteria.map((c) => ({
          id: c.id,
          text: c.text,
          status:
            c.status === "achieved" || c.status === "satisfied"
              ? "satisfied"
              : c.status === "failed"
                ? "failed"
                : c.status === "blocked"
                  ? "blocked"
                  : "pending",
          evidence: c.evidence,
          requiredIntentRevision: Math.max(1, goal.version),
          verifiedIntentRevision:
            c.status === "achieved" || c.status === "satisfied" ? Math.max(1, goal.version) : undefined,
        }));

        const lifecycle: WorkLifecycle =
          goal.status === "paused"
            ? "paused"
            : goal.status === "achieved" || goal.status === "cancelled"
              ? "closed"
              : "active";

        const closureConclusion: WorkClosureConclusion | null =
          goal.status === "achieved"
            ? "succeeded"
            : goal.status === "cancelled"
              ? "cancelled"
              : null;

        const snapshotId = `snap-${goal.id}-r${Math.max(1, goal.version)}`;
        const defaultAuth: AuthorizationSnapshotDto = {
          allowedScope: ["workspace:read", "workspace:write"],
          targetWhitelist: [],
          allowGitBranchOrWorktree: true,
          allowRedelegation: true,
        };
        const defaultBudget: WorkBudgetDto = {
          windowId: `win-${goal.id}`,
          maxTokens: 500_000,
          usedTokens: 0,
          reservedTokens: 0,
          unknownUsageCount: 0,
          windowStartAt: goal.created_at,
        };

        insertWorkStmt.run({
          workId: goal.id,
          userId: defaultUserId,
          legacyTaskId: goal.task_id,
          ownerBotId: defaultBotId,
          intentRevision: Math.max(1, goal.version),
          title: goal.title,
          goalDescription: goal.description,
          criteriaJson: JSON.stringify(upgradedCriteria),
          authorizationJson: JSON.stringify(defaultAuth),
          snapshotId,
          lifecycle,
          closureConclusion,
          progressSummary: goal.progress_summary,
          stateVersion: Math.max(1, goal.version),
          budgetJson: JSON.stringify(defaultBudget),
          createdBy: goal.created_by === "agent" ? "bot" : "user",
          createdAt: goal.created_at,
          updatedAt: goal.updated_at,
          closedAt: goal.achieved_at,
        });

        insertSnapshotStmt.run({
          snapshotId,
          workId: goal.id,
          intentRevision: Math.max(1, goal.version),
          title: goal.title,
          goalDescription: goal.description,
          criteriaJson: JSON.stringify(upgradedCriteria),
          authorizationJson: JSON.stringify(defaultAuth),
          expertSnapshotJson: JSON.stringify(defaultExpertSnapshot),
          createdBy: goal.created_by === "agent" ? "bot" : "user",
          createdAt: goal.created_at,
        });

        insertConvLinkStmt.run({
          workId: goal.id,
          conversationId: goal.task_id,
          updatedAt: goal.updated_at,
        });
      }
    }

    db.pragma(`user_version = ${BOT_WORK_PHASE1_SCHEMA_VERSION}`);
  });

  migrateTx();
}

type BotRow = {
  bot_id: string;
  user_id: string;
  display_name: string;
  responsibility_summary: string;
  expert_profile_json: string;
  memory_scope_json: string;
  foreground_allowed_tools_json: string;
  notification_policy_json: string;
  status: BotLifecycleStatus;
  created_at: number;
  updated_at: number;
};

export class BotRegistry {
  constructor(private readonly db: Database.Database) {}

  registerBot(bot: Omit<BotDto, "createdAt" | "updatedAt">): BotDto {
    const now = Date.now();
    this.db
      .prepare(`
        INSERT INTO bots (
          bot_id, user_id, display_name, responsibility_summary,
          expert_profile_json, memory_scope_json, foreground_allowed_tools_json,
          notification_policy_json, status, created_at, updated_at
        ) VALUES (
          @botId, @userId, @displayName, @responsibilitySummary,
          @expertProfileJson, @memoryScopeJson, @foregroundAllowedToolsJson,
          @notificationPolicyJson, @status, @now, @now
        )
      `)
      .run({
        botId: bot.botId,
        userId: bot.userId,
        displayName: bot.displayName,
        responsibilitySummary: bot.responsibilitySummary,
        expertProfileJson: JSON.stringify(bot.expertProfile),
        memoryScopeJson: JSON.stringify(bot.memoryScope),
        foregroundAllowedToolsJson: JSON.stringify(bot.foregroundAllowedTools),
        notificationPolicyJson: bot.notificationPolicyJson || "{}",
        status: bot.status,
        now,
      });
    return this.getBot(bot.botId)!;
  }

  getBot(botId: string): BotDto | undefined {
    const row = this.db
      .prepare("SELECT * FROM bots WHERE bot_id = ?")
      .get(botId) as BotRow | undefined;
    return row ? mapBotRow(row) : undefined;
  }

  /**
   * 绑定渠道入口到稳定 botId (满足 BW-A01: 桌面/手机/其他渠道访问同一 botId)
   */
  bindChannel(binding: BotChannelBindingDto): void {
    this.db
      .prepare(`
        INSERT INTO bot_channel_bindings (channel_type, channel_identifier, bot_id, bound_at)
        VALUES (@channelType, @channelIdentifier, @botId, @boundAt)
        ON CONFLICT(channel_type, channel_identifier)
        DO UPDATE SET bot_id = excluded.bot_id, bound_at = excluded.bound_at
      `)
      .run(binding);
  }

  resolveBotByChannel(
    channelType: BotChannelBindingDto["channelType"],
    channelIdentifier: string,
  ): BotDto | undefined {
    const row = this.db
      .prepare(`
        SELECT b.* FROM bots b
        INNER JOIN bot_channel_bindings c ON b.bot_id = c.bot_id
        WHERE c.channel_type = ? AND c.channel_identifier = ?
      `)
      .get(channelType, channelIdentifier) as BotRow | undefined;
    return row ? mapBotRow(row) : undefined;
  }

  /**
   * 专家模板更新暂存为 pendingVersion，不静默篡改在途快照 (满足 BW-A07)
   */
  stageExpertTemplateUpdate(botId: string, newVersion: number): BotDto | undefined {
    const bot = this.getBot(botId);
    if (!bot) return undefined;
    const updatedProfile: ExpertProfileSnapshotDto = {
      ...bot.expertProfile,
      desiredVersion: newVersion,
      pendingVersion: newVersion,
    };
    this.db
      .prepare("UPDATE bots SET expert_profile_json = @json, updated_at = @now WHERE bot_id = @botId")
      .run({
        json: JSON.stringify(updatedProfile),
        now: Date.now(),
        botId,
      });
    return this.getBot(botId);
  }

  /**
   * 显式采纳新专家配置版本，后续新建/修改的快照使用新版本，已在途快照保持不变 (满足 BW-A07)
   */
  adoptPendingExpertVersion(botId: string, newSystemPromptHash: string): BotDto | undefined {
    const bot = this.getBot(botId);
    if (!bot) return undefined;
    const nextVersion = bot.expertProfile.pendingVersion ?? bot.expertProfile.desiredVersion;
    const updatedProfile: ExpertProfileSnapshotDto = {
      ...bot.expertProfile,
      activeVersion: nextVersion,
      desiredVersion: nextVersion,
      pendingVersion: undefined,
      systemPromptHash: newSystemPromptHash,
      capturedAt: Date.now(),
    };
    this.db
      .prepare("UPDATE bots SET expert_profile_json = @json, updated_at = @now WHERE bot_id = @botId")
      .run({
        json: JSON.stringify(updatedProfile),
        now: Date.now(),
        botId,
      });
    return this.getBot(botId);
  }

  /**
   * 停用 Bot 守卫 (满足 BW-A06):
   * 停用后阻止新增调度，并返回当前受影响的未结束 Work 列表；严禁物理删除仍有未结束事项的 Bot
   */
  disableBot(botId: string): {
    ok: boolean;
    bot?: BotDto;
    affectedActiveWorkIds: string[];
  } {
    const bot = this.getBot(botId);
    if (!bot) return { ok: false, affectedActiveWorkIds: [] };

    const activeWorks = this.db
      .prepare("SELECT work_id FROM works WHERE owner_bot_id = ? AND lifecycle != 'closed'")
      .all(botId) as Array<{ work_id: string }>;

    this.db
      .prepare("UPDATE bots SET status = 'disabled', updated_at = ? WHERE bot_id = ?")
      .run(Date.now(), botId);

    return {
      ok: true,
      bot: this.getBot(botId),
      affectedActiveWorkIds: activeWorks.map((r) => r.work_id),
    };
  }
}

function mapBotRow(row: BotRow): BotDto {
  return {
    botId: row.bot_id,
    userId: row.user_id,
    displayName: row.display_name,
    responsibilitySummary: row.responsibility_summary,
    expertProfile: JSON.parse(row.expert_profile_json) as ExpertProfileSnapshotDto,
    memoryScope: JSON.parse(row.memory_scope_json) as BotMemoryScopeDto,
    foregroundAllowedTools: JSON.parse(row.foreground_allowed_tools_json) as string[],
    notificationPolicyJson: row.notification_policy_json,
    status: row.status,
    createdAt: row.created_at,
    updatedAt: row.updated_at,
  };
}

type ReceiptRow = {
  work_id: string;
  operation_id: string;
  operation_type: string;
  payload_hash: string;
  result_snapshot_json: string;
  outbox_event_id: string | null;
  created_at: number;
};

export class WorkRepository {
  constructor(private readonly db: Database.Database) {}

  get(workId: string): WorkDto | undefined {
    const row = this.db
      .prepare("SELECT * FROM works WHERE work_id = ?")
      .get(workId) as WorkRow | undefined;
    if (!row) return undefined;
    return this.hydrateWork(row);
  }

  listByConversation(conversationId: string): WorkDto[] {
    const rows = this.db
      .prepare(`
        SELECT w.* FROM works w
        INNER JOIN work_conversations wc ON w.work_id = wc.work_id
        WHERE wc.conversation_id = ?
        ORDER BY wc.last_referenced_at DESC, w.updated_at DESC
      `)
      .all(conversationId) as WorkRow[];
    return rows.map((r) => this.hydrateWork(r));
  }

  getLiveByLegacyTask(legacyTaskId: string): WorkDto | undefined {
    const row = this.db
      .prepare(`
        SELECT * FROM works
        WHERE legacy_task_id = ? AND lifecycle IN ('active', 'paused', 'closing')
        ORDER BY CASE WHEN lifecycle = 'active' THEN 0 ELSE 1 END, updated_at DESC
        LIMIT 1
      `)
      .get(legacyTaskId) as WorkRow | undefined;
    return row ? this.hydrateWork(row) : undefined;
  }

  linkConversation(workId: string, conversationId: string, channelType = "desktop"): void {
    this.db
      .prepare(`
        INSERT INTO work_conversations (work_id, conversation_id, channel_type, last_referenced_at)
        VALUES (@workId, @conversationId, @channelType, @now)
        ON CONFLICT(work_id, conversation_id)
        DO UPDATE SET last_referenced_at = excluded.last_referenced_at
      `)
      .run({
        workId,
        conversationId,
        channelType,
        now: Date.now(),
      });
  }

  getReceipt(workId: string, operationId: string): ReceiptRow | undefined {
    return this.db
      .prepare("SELECT * FROM work_operation_receipts WHERE work_id = ? AND operation_id = ?")
      .get(workId, operationId) as ReceiptRow | undefined;
  }

  findReceiptByOperationId(operationId: string): ReceiptRow | undefined {
    return this.db
      .prepare("SELECT * FROM work_operation_receipts WHERE operation_id = ? LIMIT 1")
      .get(operationId) as ReceiptRow | undefined;
  }

  saveReceipt(params: {
    workId: string;
    operationId: string;
    operationType: string;
    payloadHash: string;
    workSnapshot: WorkDto;
    outboxEventId?: string;
  }): void {
    this.db
      .prepare(`
        INSERT INTO work_operation_receipts (
          work_id, operation_id, operation_type, payload_hash,
          result_snapshot_json, outbox_event_id, created_at
        ) VALUES (
          @workId, @operationId, @operationType, @payloadHash,
          @resultSnapshotJson, @outboxEventId, @now
        )
      `)
      .run({
        workId: params.workId,
        operationId: params.operationId,
        operationType: params.operationType,
        payloadHash: params.payloadHash,
        resultSnapshotJson: JSON.stringify(params.workSnapshot),
        outboxEventId: params.outboxEventId ?? null,
        now: Date.now(),
      });
  }

  appendOutboxEvent(envelope: WorkEventEnvelopeDto): void {
    this.db
      .prepare(`
        INSERT INTO work_events_outbox (
          event_id, dedupe_key, event_type, work_id, execution_id,
          intent_revision, state_version, owner_epoch, request_id,
          causal_event_id, payload_json, delivery_status, attempt_count, created_at
        ) VALUES (
          @eventId, @dedupeKey, @eventType, @workId, @executionId,
          @intentRevision, @stateVersion, @ownerEpoch, @requestId,
          @causalEventId, @payloadJson, 'pending', 0, @timestamp
        )
      `)
      .run({
        eventId: envelope.eventId,
        dedupeKey: envelope.dedupeKey,
        eventType: envelope.eventType,
        workId: envelope.workId,
        executionId: envelope.executionId ?? null,
        intentRevision: envelope.intentRevision,
        stateVersion: envelope.stateVersion,
        ownerEpoch: envelope.ownerEpoch,
        requestId: envelope.requestId,
        causalEventId: envelope.causalEventId ?? null,
        payloadJson: JSON.stringify(envelope.payload),
        timestamp: envelope.timestamp,
      });
  }

  listOpenBlockers(workId: string): WorkBlockerDto[] {
    const rows = this.db
      .prepare("SELECT * FROM work_blockers WHERE work_id = ? AND status = 'open' ORDER BY created_at ASC")
      .all(workId) as Array<{
      blocker_id: string;
      work_id: string;
      reported_by_bot_id: string;
      reason: string;
      unknown_external_effect_ref: string | null;
      waiting_for_user_input: number;
      status: "open" | "cleared";
      created_at: number;
      cleared_at: number | null;
    }>;
    return rows.map((r) => ({
      blockerId: r.blocker_id,
      workId: r.work_id,
      reportedByBotId: r.reported_by_bot_id,
      reason: r.reason,
      unknownExternalEffectRef: r.unknown_external_effect_ref ?? undefined,
      waitingForUserInput: Boolean(r.waiting_for_user_input),
      status: r.status,
      createdAt: r.created_at,
      clearedAt: r.cleared_at ?? undefined,
    }));
  }

  private hydrateWork(row: WorkRow): WorkDto {
    const convRows = this.db
      .prepare("SELECT conversation_id FROM work_conversations WHERE work_id = ? ORDER BY last_referenced_at DESC")
      .all(row.work_id) as Array<{ conversation_id: string }>;

    const participantRows = this.db
      .prepare("SELECT * FROM work_participants WHERE work_id = ? ORDER BY added_at ASC")
      .all(row.work_id) as Array<{
      work_id: string;
      bot_id: string;
      role: WorkParticipantRole;
      delegated_scope_json: string;
      expert_snapshot_json: string;
      added_at: number;
    }>;

    return {
      workId: row.work_id,
      userId: row.user_id,
      projectId: row.project_id ?? undefined,
      workspacePath: row.workspace_path ?? undefined,
      sourceMessageId: row.source_message_id ?? undefined,
      legacyTaskId: row.legacy_task_id ?? undefined,
      conversationIds: convRows.map((c) => c.conversation_id),
      ownerBotId: row.owner_bot_id,
      ownerEpoch: row.owner_epoch,
      participants: participantRows.map((p) => ({
        workId: p.work_id,
        botId: p.bot_id,
        role: p.role,
        delegatedScope: JSON.parse(p.delegated_scope_json) as string[],
        expertSnapshot: JSON.parse(p.expert_snapshot_json) as ExpertProfileSnapshotDto,
        addedAt: p.added_at,
      })),
      intentRevision: row.intent_revision,
      title: row.title,
      goalDescription: row.goal_description ?? undefined,
      constraints: JSON.parse(row.constraints_json) as string[],
      criteria: JSON.parse(row.criteria_json) as WorkCriterionDto[],
      authorization: JSON.parse(row.authorization_json) as AuthorizationSnapshotDto,
      activeIntentSnapshotId: row.active_intent_snapshot_id,
      mode: row.mode,
      lifecycle: row.lifecycle,
      closureConclusion: row.closure_conclusion ?? undefined,
      closureSummary: row.closure_summary ?? undefined,
      acceptedDeliverableIds: JSON.parse(row.accepted_deliverable_ids_json) as string[],
      activeExecutionIds: JSON.parse(row.active_execution_ids_json) as string[],
      progressSummary: row.progress_summary ?? undefined,
      watchRule: row.watch_rule_json ? (JSON.parse(row.watch_rule_json) as WorkWatchRuleDto) : undefined,
      stateVersion: row.state_version,
      budget: JSON.parse(row.budget_json) as WorkBudgetDto,
      createdBy: row.created_by,
      createdAt: row.created_at,
      updatedAt: row.updated_at,
      closedAt: row.closed_at ?? undefined,
    };
  }
}

/**
 * WorkService — 事项权威控制面服务 (实现 F01 ~ F14 核心控制语义)
 */
export class WorkService {
  constructor(
    private readonly db: Database.Database,
    private readonly repo: WorkRepository,
    private readonly botRegistry: BotRegistry,
    private readonly emitUiPatch?: (work: WorkDto) => void,
  ) {}

  /**
   * 1. 创建事项 (createWork)
   * 支持 operationId 幂等校验 (BW-A05) 与不可变要求快照生成 (F06)
   */
  createWork(params: CreateWorkParams): WorkControlResult {
    const payloadHash = computeOperationPayloadHash("createWork", params);
    const existingReceipt = this.repo.findReceiptByOperationId(params.operationId);
    if (existingReceipt) {
      if (existingReceipt.payload_hash !== payloadHash) {
        return {
          ok: false,
          code: "idempotency_conflict",
          message: `operationId ${params.operationId} 已绑定不同的请求内容 (BW-A05)`,
        };
      }
      return {
        ok: true,
        replayedFromReceipt: true,
        operationId: params.operationId,
        work: JSON.parse(existingReceipt.result_snapshot_json) as WorkDto,
        outboxEventId: existingReceipt.outbox_event_id ?? undefined,
      };
    }

    if (!params.title?.trim()) {
      return { ok: false, code: "invalid_request", message: "Work 标题不能为空" };
    }

    const ownerBot = this.botRegistry.getBot(params.ownerBotId);
    if (!ownerBot) {
      return { ok: false, code: "not_found", message: `找不到负责人 Bot: ${params.ownerBotId}` };
    }
    if (ownerBot.status === "disabled") {
      return { ok: false, code: "bot_disabled", message: `Bot ${params.ownerBotId} 已停用，禁止分配新事项 (BW-A06)` };
    }

    const tx = this.db.transaction((): WorkControlResult => {
      const now = Date.now();
      const workId = `work-${randomUUID()}`;
      const snapshotId = `snap-${workId}-r1`;

      const criteria: WorkCriterionDto[] = (params.criteria ?? [])
        .map((c) => ({
          id: c.id?.trim() || `crit-${randomUUID()}`,
          text: c.text.trim().slice(0, 500),
          status: "pending" as const,
          requiredIntentRevision: 1,
          requiresRealEnvironmentCheck: c.requiresRealEnvironmentCheck,
        }))
        .filter((c) => c.text.length > 0);

      const authorization: AuthorizationSnapshotDto = {
        allowedScope: params.authorization?.allowedScope ?? ["workspace:read", "workspace:write"],
        targetWhitelist: params.authorization?.targetWhitelist ?? [],
        allowGitBranchOrWorktree: params.authorization?.allowGitBranchOrWorktree ?? true,
        budgetLimitTokens: params.authorization?.budgetLimitTokens,
        expiresAt: params.authorization?.expiresAt,
        allowRedelegation: params.authorization?.allowRedelegation ?? true,
      };

      const budget: WorkBudgetDto = {
        windowId: params.budget?.windowId ?? `win-${workId}-1`,
        maxTokens: params.budget?.maxTokens ?? 500_000,
        usedTokens: params.budget?.usedTokens ?? 0,
        reservedTokens: params.budget?.reservedTokens ?? 0,
        unknownUsageCount: params.budget?.unknownUsageCount ?? 0,
        windowStartAt: now,
        windowExpiresAt: params.budget?.windowExpiresAt,
      };

      this.db
        .prepare(`
          INSERT INTO works (
            work_id, user_id, project_id, workspace_path, source_message_id, legacy_task_id,
            owner_bot_id, owner_epoch, intent_revision, title, goal_description,
            constraints_json, criteria_json, authorization_json, active_intent_snapshot_id,
            mode, lifecycle, accepted_deliverable_ids_json, active_execution_ids_json,
            watch_rule_json, state_version, budget_json, created_by, created_at, updated_at
          ) VALUES (
            @workId, @userId, @projectId, @workspacePath, @sourceMessageId, @legacyTaskId,
            @ownerBotId, 1, 1, @title, @goalDescription,
            @constraintsJson, @criteriaJson, @authorizationJson, @snapshotId,
            @mode, 'active', '[]', '[]',
            @watchRuleJson, 1, @budgetJson, @createdBy, @now, @now
          )
        `)
        .run({
          workId,
          userId: params.userId,
          projectId: params.projectId ?? null,
          workspacePath: params.workspacePath ?? null,
          sourceMessageId: params.sourceMessageId ?? null,
          legacyTaskId: params.legacyTaskId ?? null,
          ownerBotId: params.ownerBotId,
          title: params.title.trim().slice(0, 240),
          goalDescription: params.goalDescription?.trim().slice(0, 2000) ?? null,
          constraintsJson: JSON.stringify(params.constraints ?? []),
          criteriaJson: JSON.stringify(criteria),
          authorizationJson: JSON.stringify(authorization),
          snapshotId,
          mode: params.mode ?? "once",
          watchRuleJson: params.watchRule ? JSON.stringify(params.watchRule) : null,
          budgetJson: JSON.stringify(budget),
          createdBy: params.actorUserId ? "user" : "bot",
          now,
        });

      this.db
        .prepare(`
          INSERT INTO work_intent_snapshots (
            snapshot_id, work_id, intent_revision, title, goal_description,
            constraints_json, criteria_json, authorization_json, expert_snapshot_json,
            created_by, source_message_id, created_at
          ) VALUES (
            @snapshotId, @workId, 1, @title, @goalDescription,
            @constraintsJson, @criteriaJson, @authorizationJson, @expertSnapshotJson,
            @createdBy, @sourceMessageId, @now
          )
        `)
        .run({
          snapshotId,
          workId,
          title: params.title.trim().slice(0, 240),
          goalDescription: params.goalDescription?.trim().slice(0, 2000) ?? null,
          constraintsJson: JSON.stringify(params.constraints ?? []),
          criteriaJson: JSON.stringify(criteria),
          authorizationJson: JSON.stringify(authorization),
          expertSnapshotJson: JSON.stringify(ownerBot.expertProfile),
          createdBy: params.actorUserId ? "user" : "bot",
          sourceMessageId: params.sourceMessageId ?? null,
          now,
        });

      if (params.conversationId) {
        this.repo.linkConversation(workId, params.conversationId);
      }

      const outboxEventId = `evt-${randomUUID()}`;
      this.repo.appendOutboxEvent({
        eventId: outboxEventId,
        source: params.actorUserId ? "user" : "foreground_bot",
        sourceEventId: params.sourceMessageId ?? params.operationId,
        dedupeKey: `work:${workId}:op:${params.operationId}`,
        eventType: "work.created",
        workId,
        intentRevision: 1,
        stateVersion: 1,
        ownerEpoch: 1,
        requestId: params.operationId,
        causalEventId: params.causalEventId,
        timestamp: now,
        payload: { title: params.title, mode: params.mode ?? "once" },
      });

      const createdWork = this.repo.get(workId)!;
      this.repo.saveReceipt({
        workId,
        operationId: params.operationId,
        operationType: "createWork",
        payloadHash,
        workSnapshot: createdWork,
        outboxEventId,
      });

      return {
        ok: true,
        replayedFromReceipt: false,
        operationId: params.operationId,
        work: createdWork,
        outboxEventId,
      };
    });

    const result = tx();
    if (result.ok) this.emitUiPatch?.(result.work);
    return result;
  }

  /**
   * 2. 修改事项要求 (amendWork)
   * 实质修改目标、约束、验收条件或授权时递增 intentRevision 与 stateVersion (F06, BW-A02, BW-A21, BW-A23)
   */
  amendWork(params: AmendWorkParams): WorkControlResult {
    return this.executeMutatingControl(params.workId, "amendWork", params, (current, now) => {
      if (current.lifecycle === "closed" || current.lifecycle === "closing") {
        return {
          ok: false,
          code: "invalid_lifecycle_transition",
          message: `Work 处于 ${current.lifecycle} 状态，无法直接修改要求`,
          latestWorkSnapshot: current,
        };
      }

      const nextIntentRevision = current.intentRevision + 1;
      const nextStateVersion = current.stateVersion + 1;
      const snapshotId = `snap-${current.workId}-r${nextIntentRevision}`;

      const nextTitle = params.title?.trim().slice(0, 240) || current.title;
      const nextDescription =
        params.goalDescription === undefined
          ? current.goalDescription
          : params.goalDescription.trim().slice(0, 2000) || undefined;
      const nextConstraints = params.constraints ?? current.constraints;

      const nextCriteria: WorkCriterionDto[] = params.criteria
        ? params.criteria.map((c) => ({
            id: c.id?.trim() || `crit-${randomUUID()}`,
            text: c.text.trim().slice(0, 500),
            status: c.status ?? "pending",
            evidence: c.evidence,
            requiredIntentRevision: nextIntentRevision,
            verifiedIntentRevision: c.status === "satisfied" ? nextIntentRevision : undefined,
            requiresRealEnvironmentCheck: c.requiresRealEnvironmentCheck,
          }))
        : current.criteria.map((c) => ({
            ...c,
            requiredIntentRevision: nextIntentRevision,
            // 要求实质变更后，需实机测试的验收项需对应新版本重新核验 (BW-A34)
            status: c.requiresRealEnvironmentCheck ? "pending" : c.status,
          }));

      const nextAuth: AuthorizationSnapshotDto = {
        ...current.authorization,
        ...params.authorization,
      };

      const invalidatedSet = new Set(params.invalidateExecutionIds ?? []);
      const remainingExecutions = current.activeExecutionIds.filter((id) => !invalidatedSet.has(id));

      // 将已被新要求取代的旧完成通知或提议成果标记为 superseded (BW-A23, BW-A28)
      this.db
        .prepare(`
          UPDATE work_deliverables
          SET status = 'superseded'
          WHERE work_id = @workId AND status = 'proposed' AND intent_revision < @nextIntentRevision
        `)
        .run({ workId: current.workId, nextIntentRevision });

      const ownerBot = this.botRegistry.getBot(current.ownerBotId)!;
      this.db
        .prepare(`
          INSERT INTO work_intent_snapshots (
            snapshot_id, work_id, intent_revision, title, goal_description,
            constraints_json, criteria_json, authorization_json, expert_snapshot_json,
            created_by, source_message_id, created_at
          ) VALUES (
            @snapshotId, @workId, @intentRevision, @title, @goalDescription,
            @constraintsJson, @criteriaJson, @authorizationJson, @expertSnapshotJson,
            @createdBy, @sourceMessageId, @now
          )
        `)
        .run({
          snapshotId,
          workId: current.workId,
          intentRevision: nextIntentRevision,
          title: nextTitle,
          goalDescription: nextDescription ?? null,
          constraintsJson: JSON.stringify(nextConstraints),
          criteriaJson: JSON.stringify(nextCriteria),
          authorizationJson: JSON.stringify(nextAuth),
          expertSnapshotJson: JSON.stringify(ownerBot.expertProfile),
          createdBy: params.actorUserId ? "user" : "bot",
          sourceMessageId: params.sourceMessageId ?? null,
          now,
        });

      this.db
        .prepare(`
          UPDATE works SET
            intent_revision = @intentRevision,
            title = @title,
            goal_description = @goalDescription,
            constraints_json = @constraintsJson,
            criteria_json = @criteriaJson,
            authorization_json = @authorizationJson,
            active_intent_snapshot_id = @snapshotId,
            active_execution_ids_json = @activeExecutionsJson,
            watch_rule_json = @watchRuleJson,
            state_version = @stateVersion,
            updated_at = @now
          WHERE work_id = @workId
        `)
        .run({
          workId: current.workId,
          intentRevision: nextIntentRevision,
          title: nextTitle,
          goalDescription: nextDescription ?? null,
          constraintsJson: JSON.stringify(nextConstraints),
          criteriaJson: JSON.stringify(nextCriteria),
          authorizationJson: JSON.stringify(nextAuth),
          snapshotId,
          activeExecutionsJson: JSON.stringify(remainingExecutions),
          watchRuleJson: params.watchRule
            ? JSON.stringify(params.watchRule)
            : current.watchRule
              ? JSON.stringify(current.watchRule)
              : null,
          stateVersion: nextStateVersion,
          now,
        });

      if (params.conversationId) {
        this.repo.linkConversation(current.workId, params.conversationId);
      }

      const outboxEventId = `evt-${randomUUID()}`;
      this.repo.appendOutboxEvent({
        eventId: outboxEventId,
        source: params.actorUserId ? "user" : "foreground_bot",
        sourceEventId: params.sourceMessageId ?? params.operationId,
        dedupeKey: `work:${current.workId}:op:${params.operationId}`,
        eventType: "work.amended",
        workId: current.workId,
        intentRevision: nextIntentRevision,
        stateVersion: nextStateVersion,
        ownerEpoch: current.ownerEpoch,
        requestId: params.operationId,
        causalEventId: params.causalEventId,
        timestamp: now,
        payload: {
          previousRevision: current.intentRevision,
          nextRevision: nextIntentRevision,
          invalidatedExecutions: Array.from(invalidatedSet),
        },
      });

      return { ok: true, outboxEventId };
    });
  }

  /**
   * 3. 暂停事项 (pauseWork) — active -> paused (阻止新动作，请求在途执行安全等待, F08 & BW-A26)
   */
  pauseWork(params: PauseWorkParams): WorkControlResult {
    return this.executeMutatingControl(params.workId, "pauseWork", params, (current, now) => {
      if (current.lifecycle !== "active") {
        return {
          ok: false,
          code: "invalid_lifecycle_transition",
          message: `仅 active 事项可暂停，当前状态为 ${current.lifecycle}`,
          latestWorkSnapshot: current,
        };
      }
      const nextStateVersion = current.stateVersion + 1;
      this.db
        .prepare("UPDATE works SET lifecycle = 'paused', state_version = @stateVersion, updated_at = @now WHERE work_id = @workId")
        .run({ workId: current.workId, stateVersion: nextStateVersion, now });

      const outboxEventId = `evt-${randomUUID()}`;
      this.repo.appendOutboxEvent({
        eventId: outboxEventId,
        source: params.actorUserId ? "user" : "foreground_bot",
        sourceEventId: params.operationId,
        dedupeKey: `work:${current.workId}:op:${params.operationId}`,
        eventType: "work.pause_requested",
        workId: current.workId,
        intentRevision: current.intentRevision,
        stateVersion: nextStateVersion,
        ownerEpoch: current.ownerEpoch,
        requestId: params.operationId,
        causalEventId: params.causalEventId,
        timestamp: now,
        payload: { reason: params.reason, activeExecutionIds: current.activeExecutionIds },
      });
      return { ok: true, outboxEventId };
    });
  }

  /**
   * 4. 恢复事项 (resumeWork) — paused -> active (BW-A26)
   */
  resumeWork(params: ResumeWorkParams): WorkControlResult {
    return this.executeMutatingControl(params.workId, "resumeWork", params, (current, now) => {
      if (current.lifecycle !== "paused") {
        return {
          ok: false,
          code: "invalid_lifecycle_transition",
          message: `仅 paused 事项可恢复，当前状态为 ${current.lifecycle}`,
          latestWorkSnapshot: current,
        };
      }
      const nextStateVersion = current.stateVersion + 1;
      this.db
        .prepare("UPDATE works SET lifecycle = 'active', state_version = @stateVersion, updated_at = @now WHERE work_id = @workId")
        .run({ workId: current.workId, stateVersion: nextStateVersion, now });

      const outboxEventId = `evt-${randomUUID()}`;
      this.repo.appendOutboxEvent({
        eventId: outboxEventId,
        source: params.actorUserId ? "user" : "foreground_bot",
        sourceEventId: params.operationId,
        dedupeKey: `work:${current.workId}:op:${params.operationId}`,
        eventType: "work.resumed",
        workId: current.workId,
        intentRevision: current.intentRevision,
        stateVersion: nextStateVersion,
        ownerEpoch: current.ownerEpoch,
        requestId: params.operationId,
        causalEventId: params.causalEventId,
        timestamp: now,
        payload: {},
      });
      return { ok: true, outboxEventId };
    });
  }

  /**
   * 5. 请求取消事项 (cancelWork)
   * 若存在在途执行或未结算副作用，先进入 closing 发停止/回执核对指令；否则直接 closed(cancelled) (F08 & BW-A24)
   */
  cancelWork(params: CancelWorkParams): WorkControlResult {
    return this.executeMutatingControl(params.workId, "cancelWork", params, (current, now) => {
      if (current.lifecycle === "closed") {
        return {
          ok: false,
          code: "invalid_lifecycle_transition",
          message: "事项已关闭",
          latestWorkSnapshot: current,
        };
      }
      const hasInFlight =
        Boolean(params.hasInFlightSideEffects) || current.activeExecutionIds.length > 0;
      const targetLifecycle: WorkLifecycle = hasInFlight ? "closing" : "closed";
      const conclusion: WorkClosureConclusion | null = hasInFlight ? null : "cancelled";
      const nextStateVersion = current.stateVersion + 1;

      this.db
        .prepare(`
          UPDATE works SET
            lifecycle = @lifecycle,
            closure_conclusion = COALESCE(@conclusion, closure_conclusion),
            closure_summary = COALESCE(@reason, closure_summary),
            state_version = @stateVersion,
            updated_at = @now,
            closed_at = CASE WHEN @lifecycle = 'closed' THEN @now ELSE closed_at END
          WHERE work_id = @workId
        `)
        .run({
          workId: current.workId,
          lifecycle: targetLifecycle,
          conclusion,
          reason: params.reason ?? null,
          stateVersion: nextStateVersion,
          now,
        });

      const outboxEventId = `evt-${randomUUID()}`;
      this.repo.appendOutboxEvent({
        eventId: outboxEventId,
        source: params.actorUserId ? "user" : "foreground_bot",
        sourceEventId: params.operationId,
        dedupeKey: `work:${current.workId}:op:${params.operationId}`,
        eventType: hasInFlight ? "work.cancel_settling" : "work.cancelled",
        workId: current.workId,
        intentRevision: current.intentRevision,
        stateVersion: nextStateVersion,
        ownerEpoch: current.ownerEpoch,
        requestId: params.operationId,
        causalEventId: params.causalEventId,
        timestamp: now,
        payload: {
          reason: params.reason,
          activeExecutionIdsToStop: current.activeExecutionIds,
        },
      });
      return { ok: true, outboxEventId };
    });
  }

  /**
   * 6. 委派协作者或原子交接主负责人 (delegateWork)
   * 当 transferOwnership=true 时，在同一事务中更新 ownerBotId 并递增 ownerEpoch，不递增 intentRevision (F04, F06, F11 & BW-A27)
   */
  delegateWork(params: DelegateWorkParams): WorkControlResult {
    return this.executeMutatingControl(params.workId, "delegateWork", params, (current, now) => {
      const targetBot = this.botRegistry.getBot(params.targetBotId);
      if (!targetBot) {
        return {
          ok: false,
          code: "not_found",
          message: `目标 Bot 不存在: ${params.targetBotId}`,
          latestWorkSnapshot: current,
        };
      }
      if (targetBot.status === "disabled") {
        return {
          ok: false,
          code: "bot_disabled",
          message: `目标 Bot ${params.targetBotId} 已停用`,
          latestWorkSnapshot: current,
        };
      }

      const nextStateVersion = current.stateVersion + 1;
      const nextOwnerEpoch = params.transferOwnership ? current.ownerEpoch + 1 : current.ownerEpoch;
      const nextOwnerBotId = params.transferOwnership ? params.targetBotId : current.ownerBotId;

      if (params.transferOwnership) {
        this.db
          .prepare(`
            UPDATE works SET
              owner_bot_id = @ownerBotId,
              owner_epoch = @ownerEpoch,
              state_version = @stateVersion,
              updated_at = @now
            WHERE work_id = @workId
          `)
          .run({
            workId: current.workId,
            ownerBotId: nextOwnerBotId,
            ownerEpoch: nextOwnerEpoch,
            stateVersion: nextStateVersion,
            now,
          });
      } else {
        this.db
          .prepare(`
            INSERT INTO work_participants (work_id, bot_id, role, delegated_scope_json, expert_snapshot_json, added_at)
            VALUES (@workId, @botId, @role, @scopeJson, @expertSnapshotJson, @now)
            ON CONFLICT(work_id, bot_id) DO UPDATE SET
              role = excluded.role,
              delegated_scope_json = excluded.delegated_scope_json
          `)
          .run({
            workId: current.workId,
            botId: params.targetBotId,
            role: params.role ?? "worker",
            scopeJson: JSON.stringify(params.delegatedScope ?? []),
            expertSnapshotJson: JSON.stringify(targetBot.expertProfile),
            now,
          });

        this.db
          .prepare("UPDATE works SET state_version = @stateVersion, updated_at = @now WHERE work_id = @workId")
          .run({ workId: current.workId, stateVersion: nextStateVersion, now });
      }

      const outboxEventId = `evt-${randomUUID()}`;
      this.repo.appendOutboxEvent({
        eventId: outboxEventId,
        source: "foreground_bot",
        sourceEventId: params.operationId,
        dedupeKey: `work:${current.workId}:op:${params.operationId}`,
        eventType: params.transferOwnership ? "work.owner_handed_off" : "work.participant_delegated",
        workId: current.workId,
        intentRevision: current.intentRevision,
        stateVersion: nextStateVersion,
        ownerEpoch: nextOwnerEpoch,
        requestId: params.operationId,
        causalEventId: params.causalEventId,
        timestamp: now,
        payload: {
          previousOwnerBotId: current.ownerBotId,
          newOwnerBotId: nextOwnerBotId,
          previousEpoch: current.ownerEpoch,
          newEpoch: nextOwnerEpoch,
          targetBotId: params.targetBotId,
        },
      });
      return { ok: true, outboxEventId };
    });
  }

  /**
   * 7. 提交待验收成果 (submitDeliverable — 供 worker / 后台协调者提交产出)
   * 若要求已更新 (deliverable.intentRevision < current.intentRevision)，仅归档为历史版本证据，不覆盖当前目标 (BW-A23)
   */
  submitDeliverable(params: SubmitDeliverableParams): WorkControlResult {
    return this.executeMutatingControl(
      params.workId,
      "submitDeliverable",
      params,
      (current, now) => {
        const deliverableId = params.deliverableId ?? `deliv-${randomUUID()}`;
        const submittedRevision =
          params.versionGuard?.expectedIntentRevision ?? current.intentRevision;
        const isStaleRevision = submittedRevision < current.intentRevision;

        this.db
          .prepare(`
            INSERT INTO work_deliverables (
              deliverable_id, work_id, producer_bot_id, execution_id,
              intent_revision, owner_epoch, status, summary,
              artifact_refs_json, evidence_refs_json, uncovered_scope_json,
              is_speculation, has_write_conflicts, created_at
            ) VALUES (
              @deliverableId, @workId, @producerBotId, @executionId,
              @intentRevision, @ownerEpoch, @status, @summary,
              @artifactRefsJson, @evidenceRefsJson, @uncoveredScopeJson,
              @isSpeculation, @hasWriteConflicts, @now
            )
          `)
          .run({
            deliverableId,
            workId: current.workId,
            producerBotId: params.actorBotId,
            executionId: params.executionId ?? null,
            intentRevision: submittedRevision,
            ownerEpoch: current.ownerEpoch,
            status: isStaleRevision ? "superseded" : "proposed",
            summary: params.summary,
            artifactRefsJson: JSON.stringify(params.artifactRefs ?? []),
            evidenceRefsJson: JSON.stringify(params.evidenceRefs ?? []),
            uncoveredScopeJson: JSON.stringify(params.uncoveredScope ?? []),
            isSpeculation: params.isSpeculation ? 1 : 0,
            hasWriteConflicts: params.hasWriteConflicts ? 1 : 0,
            now,
          });

        const nextStateVersion = current.stateVersion + 1;
        this.db
          .prepare("UPDATE works SET state_version = @stateVersion, updated_at = @now WHERE work_id = @workId")
          .run({ workId: current.workId, stateVersion: nextStateVersion, now });

        return { ok: true };
      },
      { allowWorkerActor: true, skipStrictRevisionMatch: true },
    );
  }

  /**
   * 8. 主负责人验收成果 (acceptDeliverable)
   * 仅当前 ownerBotId 可执行 (F04)；严禁将未经确认的推测 (isSpeculation) 或旧要求版本成果直接接纳为当前结论 (BW-A08, BW-A23, BW-A27)
   */
  acceptDeliverable(params: AcceptDeliverableParams): WorkControlResult {
    return this.executeMutatingControl(params.workId, "acceptDeliverable", params, (current, now) => {
      if (params.actorBotId !== current.ownerBotId) {
        return {
          ok: false,
          code: "unauthorized_actor",
          message: `仅当前主负责人 (${current.ownerBotId}) 可验收成果，执行者 (${params.actorBotId}) 不得自行宣布完成 (F04)`,
          latestWorkSnapshot: current,
        };
      }

      const deliv = this.db
        .prepare("SELECT * FROM work_deliverables WHERE deliverable_id = ? AND work_id = ?")
        .get(params.deliverableId, current.workId) as
        | {
            deliverable_id: string;
            intent_revision: number;
            is_speculation: number;
            has_write_conflicts: number;
          }
        | undefined;

      if (!deliv) {
        return {
          ok: false,
          code: "not_found",
          message: `找不到成果记录: ${params.deliverableId}`,
          latestWorkSnapshot: current,
        };
      }

      if (deliv.is_speculation) {
        return {
          ok: false,
          code: "speculation_not_acceptable",
          message: "未经确认的推测不能自动成为已确认事实 (BW-A08)",
          latestWorkSnapshot: current,
        };
      }

      if (deliv.intent_revision !== current.intentRevision) {
        return {
          ok: false,
          code: "version_conflict",
          message: `成果对应的要求版本 (${deliv.intent_revision}) 与当前要求版本 (${current.intentRevision}) 不一致，不能覆盖当前目标结论 (BW-A23)`,
          latestWorkSnapshot: current,
        };
      }

      const updatesMap = new Map(
        (params.criterionUpdates ?? []).map((u) => [u.criterionId, u]),
      );

      for (const criterion of current.criteria) {
        const patch = updatesMap.get(criterion.id);
        if (
          patch?.status === "satisfied" &&
          criterion.requiresRealEnvironmentCheck &&
          !patch.verifiedByRealEnvironment
        ) {
          return {
            ok: false,
            code: "unmet_acceptance_criteria",
            message: `验收条件 "${criterion.text}" 要求真实环境实测反馈，不能由自动测试或模型猜测代替 (BW-A34)`,
            latestWorkSnapshot: current,
          };
        }
      }

      const updatedCriteria = current.criteria.map((c) => {
        const patch = updatesMap.get(c.id);
        if (!patch) return c;
        return {
          ...c,
          status: patch.status,
          evidence: patch.evidence ?? c.evidence,
          verifiedIntentRevision:
            patch.status === "satisfied" ? current.intentRevision : c.verifiedIntentRevision,
        };
      });

      this.db
        .prepare(`
          UPDATE work_deliverables
          SET status = 'accepted', reviewed_at = @now, reviewed_by_bot_id = @reviewer
          WHERE deliverable_id = @deliverableId
        `)
        .run({
          deliverableId: params.deliverableId,
          now,
          reviewer: params.actorBotId,
        });

      const acceptedIds = Array.from(
        new Set([...current.acceptedDeliverableIds, params.deliverableId]),
      );
      const nextStateVersion = current.stateVersion + 1;

      this.db
        .prepare(`
          UPDATE works SET
            criteria_json = @criteriaJson,
            accepted_deliverable_ids_json = @acceptedIdsJson,
            progress_summary = COALESCE(@progressSummary, progress_summary),
            state_version = @stateVersion,
            updated_at = @now
          WHERE work_id = @workId
        `)
        .run({
          workId: current.workId,
          criteriaJson: JSON.stringify(updatedCriteria),
          acceptedIdsJson: JSON.stringify(acceptedIds),
          progressSummary: params.progressSummary ?? null,
          stateVersion: nextStateVersion,
          now,
        });

      const outboxEventId = `evt-${randomUUID()}`;
      this.repo.appendOutboxEvent({
        eventId: outboxEventId,
        source: "foreground_bot",
        sourceEventId: params.operationId,
        dedupeKey: `work:${current.workId}:op:${params.operationId}`,
        eventType: "work.deliverable_accepted",
        workId: current.workId,
        intentRevision: current.intentRevision,
        stateVersion: nextStateVersion,
        ownerEpoch: current.ownerEpoch,
        requestId: params.operationId,
        causalEventId: params.causalEventId,
        timestamp: now,
        payload: { deliverableId: params.deliverableId },
      });

      return { ok: true, outboxEventId };
    });
  }

  /**
   * 9. 拒绝成果 (rejectDeliverable)
   */
  rejectDeliverable(params: RejectDeliverableParams): WorkControlResult {
    return this.executeMutatingControl(params.workId, "rejectDeliverable", params, (current, now) => {
      if (params.actorBotId !== current.ownerBotId) {
        return {
          ok: false,
          code: "unauthorized_actor",
          message: "仅主负责人可拒绝成果",
          latestWorkSnapshot: current,
        };
      }
      this.db
        .prepare("UPDATE work_deliverables SET status = 'rejected', reviewed_at = @now, reviewed_by_bot_id = @botId WHERE deliverable_id = @id AND work_id = @workId")
        .run({
          id: params.deliverableId,
          workId: current.workId,
          botId: params.actorBotId,
          now,
        });
      const nextStateVersion = current.stateVersion + 1;
      this.db
        .prepare("UPDATE works SET state_version = @stateVersion, updated_at = @now WHERE work_id = @workId")
        .run({ workId: current.workId, stateVersion: nextStateVersion, now });
      return { ok: true };
    });
  }

  /**
   * 10. 上报阻碍或未知外部副作用 (reportBlocker — 对应 BW-A19, BW-A33)
   */
  reportBlocker(params: ReportBlockerParams): WorkControlResult {
    return this.executeMutatingControl(
      params.workId,
      "reportBlocker",
      params,
      (current, now) => {
        const blockerId = `blk-${randomUUID()}`;
        this.db
          .prepare(`
            INSERT INTO work_blockers (
              blocker_id, work_id, reported_by_bot_id, reason,
              unknown_external_effect_ref, waiting_for_user_input, status, created_at
            ) VALUES (
              @blockerId, @workId, @botId, @reason,
              @unknownRef, @waitingForUser, 'open', @now
            )
          `)
          .run({
            blockerId,
            workId: current.workId,
            botId: params.actorBotId,
            reason: params.reason,
            unknownRef: params.unknownExternalEffectRef ?? null,
            waitingForUser: params.waitingForUserInput ? 1 : 0,
            now,
          });

        const nextStateVersion = current.stateVersion + 1;
        this.db
          .prepare("UPDATE works SET state_version = @stateVersion, updated_at = @now WHERE work_id = @workId")
          .run({ workId: current.workId, stateVersion: nextStateVersion, now });
        return { ok: true };
      },
      { allowWorkerActor: true },
    );
  }

  /**
   * 11. 解除阻碍 (clearBlocker)
   */
  clearBlocker(params: ClearBlockerParams): WorkControlResult {
    return this.executeMutatingControl(params.workId, "clearBlocker", params, (current, now) => {
      this.db
        .prepare("UPDATE work_blockers SET status = 'cleared', cleared_at = @now WHERE blocker_id = @blockerId AND work_id = @workId")
        .run({ blockerId: params.blockerId, workId: current.workId, now });

      const nextStateVersion = current.stateVersion + 1;
      this.db
        .prepare("UPDATE works SET state_version = @stateVersion, updated_at = @now WHERE work_id = @workId")
        .run({ workId: current.workId, stateVersion: nextStateVersion, now });
      return { ok: true };
    });
  }

  /**
   * 12. 正常或异常关闭事项 (closeWork)
   * 严格核验:
   * - 仅主负责人或用户可关闭 (F04)
   * - 若结论为 succeeded，必须所有 criteria 已在当前 intentRevision 下满足且无未解除 blocker (BW-A04, BW-A33, BW-A34)
   */
  closeWork(params: CloseWorkParams): WorkControlResult {
    return this.executeMutatingControl(params.workId, "closeWork", params, (current, now) => {
      if (!params.actorUserId && params.actorBotId !== current.ownerBotId) {
        return {
          ok: false,
          code: "unauthorized_actor",
          message: "执行者不得自行宣布整件事完成，仅当前主负责人或用户可关闭 Work (F04)",
          latestWorkSnapshot: current,
        };
      }

      if (params.conclusion === "succeeded") {
        const unmetCriterion = current.criteria.find(
          (c) =>
            c.status !== "satisfied" ||
            c.verifiedIntentRevision !== current.intentRevision,
        );
        if (unmetCriterion) {
          return {
            ok: false,
            code: "unmet_acceptance_criteria",
            message: `验收条件尚未在当前要求版本 (r${current.intentRevision}) 下满足: "${unmetCriterion.text}" (BW-A33, BW-A34)`,
            latestWorkSnapshot: current,
          };
        }

        const openBlockers = this.repo.listOpenBlockers(current.workId);
        if (openBlockers.length > 0) {
          return {
            ok: false,
            code: "unmet_acceptance_criteria",
            message: `仍存在 ${openBlockers.length} 个未解除阻碍 (${openBlockers[0].reason})，不可宣布 succeeded`,
            latestWorkSnapshot: current,
          };
        }
      }

      const nextStateVersion = current.stateVersion + 1;
      // 关闭有限跟进 Work 时自动清空专属活跃订阅，不影响其他 watch Work (BW-A04)
      const clearedWatchRule = current.watchRule
        ? { ...current.watchRule, subscriptions: [], nextWakeAt: undefined }
        : undefined;

      this.db
        .prepare(`
          UPDATE works SET
            lifecycle = 'closed',
            closure_conclusion = @conclusion,
            closure_summary = @summary,
            watch_rule_json = @watchRuleJson,
            state_version = @stateVersion,
            updated_at = @now,
            closed_at = @now
          WHERE work_id = @workId
        `)
        .run({
          workId: current.workId,
          conclusion: params.conclusion,
          summary: params.closureSummary ?? null,
          watchRuleJson: clearedWatchRule ? JSON.stringify(clearedWatchRule) : null,
          stateVersion: nextStateVersion,
          now,
        });

      const outboxEventId = `evt-${randomUUID()}`;
      this.repo.appendOutboxEvent({
        eventId: outboxEventId,
        source: params.actorUserId ? "user" : "foreground_bot",
        sourceEventId: params.operationId,
        dedupeKey: `work:${current.workId}:op:${params.operationId}`,
        eventType: "work.closed",
        workId: current.workId,
        intentRevision: current.intentRevision,
        stateVersion: nextStateVersion,
        ownerEpoch: current.ownerEpoch,
        requestId: params.operationId,
        causalEventId: params.causalEventId,
        timestamp: now,
        payload: {
          conclusion: params.conclusion,
          closureSummary: params.closureSummary,
        },
      });

      return { ok: true, outboxEventId };
    });
  }

  /**
   * 高频流式进展/Token/速度/心跳写入 (对应 BW-A22)
   * 写入独立表 work_telemetry_state，绝不递增 works.intent_revision 或 works.state_version，保证控制操作不被高频遥测饿死
   */
  recordTelemetryProgress(params: {
    workId: string;
    executionId?: string;
    streamingText?: string;
    tokensPerSecond?: number;
  }): void {
    const now = Date.now();
    this.db
      .prepare(`
        INSERT INTO work_telemetry_state (
          work_id, active_execution_id, streaming_text, tokens_per_second, last_heartbeat_at, updated_at
        ) VALUES (
          @workId, @executionId, @streamingText, @tps, @now, @now
        )
        ON CONFLICT(work_id) DO UPDATE SET
          active_execution_id = COALESCE(excluded.active_execution_id, work_telemetry_state.active_execution_id),
          streaming_text = COALESCE(excluded.streaming_text, work_telemetry_state.streaming_text),
          tokens_per_second = COALESCE(excluded.tokens_per_second, work_telemetry_state.tokens_per_second),
          last_heartbeat_at = excluded.last_heartbeat_at,
          updated_at = excluded.updated_at
      `)
      .run({
        workId: params.workId,
        executionId: params.executionId ?? null,
        streamingText: params.streamingText ?? null,
        tps: params.tokensPerSecond ?? null,
        now,
      });
  }

  /**
   * 统一事务执行器：封装幂等回执核对 + 三元组乐观锁 (ownerEpoch / intentRevision / stateVersion)
   */
  private executeMutatingControl<TParams extends BaseControlParams>(
    workId: string,
    operationType: string,
    params: TParams,
    mutator: (
      current: WorkDto,
      now: number,
    ) =>
      | { ok: true; outboxEventId?: string }
      | { ok: false; code: WorkControlErrorCode; message: string; latestWorkSnapshot?: WorkDto },
    options?: {
      allowWorkerActor?: boolean;
      skipStrictRevisionMatch?: boolean;
    },
  ): WorkControlResult {
    const payloadHash = computeOperationPayloadHash(operationType, params);

    const tx = this.db.transaction((): WorkControlResult => {
      // 1. 幂等回执核对 (BW-A05)
      const existingReceipt = this.repo.getReceipt(workId, params.operationId);
      if (existingReceipt) {
        if (existingReceipt.payload_hash !== payloadHash) {
          return {
            ok: false,
            code: "idempotency_conflict",
            message: `operationId ${params.operationId} 已存在但请求参数不一致 (BW-A05)`,
            latestWorkSnapshot: this.repo.get(workId),
          };
        }
        return {
          ok: true,
          replayedFromReceipt: true,
          operationId: params.operationId,
          work: JSON.parse(existingReceipt.result_snapshot_json) as WorkDto,
          outboxEventId: existingReceipt.outbox_event_id ?? undefined,
        };
      }

      const current = this.repo.get(workId);
      if (!current) {
        return { ok: false, code: "not_found", message: `找不到 Work: ${workId}` };
      }

      // 2. 校验负责人代次 ownerEpoch (BW-A27: 交接后旧负责人控制决定自动失效)
      const guard = params.versionGuard;
      if (
        guard?.expectedOwnerEpoch !== undefined &&
        guard.expectedOwnerEpoch !== current.ownerEpoch
      ) {
        return {
          ok: false,
          code: "epoch_mismatch",
          message: `负责人代次已变更 (expected=${guard.expectedOwnerEpoch}, actual=${current.ownerEpoch})，旧负责人决定已失效 (BW-A27)`,
          latestWorkSnapshot: current,
        };
      }

      // 3. 校验要求版本 intentRevision (BW-A21: 用户修改要求后拒绝基于旧快照的模型决定)
      if (
        !options?.skipStrictRevisionMatch &&
        guard?.expectedIntentRevision !== undefined &&
        guard.expectedIntentRevision !== current.intentRevision
      ) {
        return {
          ok: false,
          code: "version_conflict",
          message: `要求版本已更新 (expected=${guard.expectedIntentRevision}, actual=${current.intentRevision})，请基于最新要求重新评估 (BW-A21)`,
          latestWorkSnapshot: current,
        };
      }

      // 4. 校验并发状态版本 stateVersion (F06)
      if (
        guard?.expectedStateVersion !== undefined &&
        guard.expectedStateVersion !== current.stateVersion
      ) {
        return {
          ok: false,
          code: "version_conflict",
          message: `状态版本冲突 (expected=${guard.expectedStateVersion}, actual=${current.stateVersion})`,
          latestWorkSnapshot: current,
        };
      }

      // 5. 校验操作主体授权
      if (!params.actorUserId && !options?.allowWorkerActor && params.actorBotId !== current.ownerBotId) {
        return {
          ok: false,
          code: "unauthorized_actor",
          message: `Bot ${params.actorBotId} 不是当前事项主负责人 (${current.ownerBotId})`,
          latestWorkSnapshot: current,
        };
      }

      const now = Date.now();
      const mutationOutcome = mutator(current, now);
      if (!mutationOutcome.ok) {
        return mutationOutcome;
      }

      const updatedWork = this.repo.get(workId)!;
      this.repo.saveReceipt({
        workId,
        operationId: params.operationId,
        operationType,
        payloadHash,
        workSnapshot: updatedWork,
        outboxEventId: mutationOutcome.outboxEventId,
      });

      return {
        ok: true,
        replayedFromReceipt: false,
        operationId: params.operationId,
        work: updatedWork,
        outboxEventId: mutationOutcome.outboxEventId,
      };
    });

    const result = tx();
    if (result.ok) this.emitUiPatch?.(result.work);
    return result;
  }
}

/**
 * 计算幂等操作请求体的稳定 SHA-256 摘要 (排除与业务语义无关的瞬态字段)
 */
function computeOperationPayloadHash(operationType: string, payload: unknown): string {
  const normalized = JSON.stringify({ operationType, payload }, Object.keys(payload as object).sort());
  return createHash("sha256").update(normalized).digest("hex");
}
